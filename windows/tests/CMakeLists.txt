cmake_minimum_required(VERSION 3.19)
project(vitalis_onnx_tests LANGUAGES CXX)
set(BINARY_NAME vitalis_onnx_tests)
if(NOT VITALIS_TASK_FIXTURES)
  message(FATAL_ERROR "VITALIS_TASK_FIXTURES must identify generated reference fixtures")
endif()
add_subdirectory(../onnxruntime onnxruntime)
add_executable(${BINARY_NAME} onnx_task_engine_test.cpp ../runner/onnx_task_engine.cpp)
target_compile_features(${BINARY_NAME} PRIVATE cxx_std_17)
target_compile_options(${BINARY_NAME} PRIVATE /EHsc /W4 /WX /wd4100)
target_compile_definitions(${BINARY_NAME} PRIVATE NOMINMAX _HAS_EXCEPTIONS=1)
target_link_libraries(${BINARY_NAME} PRIVATE vitalis_onnx bcrypt windowscodecs ole32)
add_custom_command(TARGET ${BINARY_NAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "$<TARGET_FILE:vitalis_onnx>" "$<TARGET_FILE_DIR:${BINARY_NAME}>")
enable_testing()
add_test(NAME windows_algorithms COMMAND ${BINARY_NAME} "${VITALIS_TASK_FIXTURES}")
